   448→function useOpenStatus() {
   449→  const [status, setStatus] = useState<{ open: boolean; note: string } | null>(null);
   450→  useEffect(() => {
   451→    const compute = () => {
   452→      try {
   453→        const now = new Date(new Date().toLocaleString("en-US", { timeZone: "Asia/Jakarta" }));
   454→        const day = now.getDay(); // 0 Minggu
   455→        const hour = now.getHours() + now.getMinutes() / 60;
   456→        const weekend = day === 0 || day === 6;
   457→        const openAt = 8;
   458→        const closeAt = weekend ? 21 : 20;
   459→        const open = hour >= openAt && hour < closeAt;
   460→        setStatus({
   461→          open,
   462→          note: open
   463→            ? `Buka sekarang — tutup pukul ${String(closeAt).padStart(2, "0")}.00 WIB`
   464→            : `Tutup sekarang — buka pukul 08.00 WIB (±${weekend ? "tutup 21.00" : "tutup 20.00"} WIB)`,
   465→        });
   466→      } catch {
   467→        setStatus({ open: true, note: "Buka setiap hari mulai 08.00 WIB" });
   468→      }
   469→    };
   470→    compute();
   471→    const t = setInterval(compute, 60000);
   472→    return () => clearInterval(t);
   473→  }, []);
   474→  return status;
   475→}
